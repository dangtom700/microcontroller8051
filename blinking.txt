ORG 000H
	
	LOOP:
		MOV A, #001H
		MOV P2, A
		MOV P1, #00110000B;# - - - C D - - -
		ACALL DELAY;
		
		RL A;
		MOV P2, A
		MOV P1, #00100100B;
		ACALL DELAY;
		
		RL A;
		MOV P2, A
		MOV P1, #11111001B;
		ACALL DELAY;
		JMP LOOP
		
	DELAY:
		MOV R1, #20
		DELAY_OUTER:
		MOV R2, #20
		DELAY_INNER:
		NOP
		DJNZ R2, DELAY_INNER;
		DJNZ R1, DELAY_OUTER;
		RET
END
